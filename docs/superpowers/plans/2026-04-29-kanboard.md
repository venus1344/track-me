# Kanboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-stack kanban project tracker (Kanboard) for managing client projects with two-level kanban boards, blocker tracking, multi-channel notifications, and customer share links.

**Architecture:** Monorepo with React/Vite frontend and Express/TypeScript backend sharing a SQLite database. Backend exposes REST API consumed by frontend. Two phases: Phase 1 builds a fully testable backend API, Phase 2 builds the frontend UI on top.

**Tech Stack:** TypeScript, React 18, Vite 5, Tailwind CSS, React Router v6, TanStack Query v5, Express 4, better-sqlite3, express-session, bcryptjs, Resend, node-cron, Multer

```ts
import express from 'express';
import session from 'express-session';
import cors from 'cors';
import path from 'path';
import { fileURLToPath } from 'url';
import Database from 'better-sqlite3';
import { authRoutes } from './routes/auth.js';
import { requireAuth } from './middleware/auth.js';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export function createApp(db: Database.Database, sessionSecret: string) {
  const app = express();

  app.use(cors({ origin: true, credentials: true }));
  app.use(express.json());
  app.use(
    session({
      secret: sessionSecret,
      resave: false,
      saveUninitialized: false,
      cookie: { httpOnly: true, maxAge: 7 * 24 * 60 * 60 * 1000 },
    })
  );

  // Serve uploaded files
  app.use('/uploads', express.static(path.join(__dirname, '../../uploads')));

  // Health check
  app.get('/api/health', (_req, res) => {
    res.json({ status: 'ok' });
  });

  // Auth (no guard)
  app.use('/api/auth', authRoutes(db));

  return app;
}
```

```ts
import Database from 'better-sqlite3';
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

let db: Database.Database;

export function getDb(dbPath?: string): Database.Database {
  if (db) return db;

  const resolvedPath = dbPath || path.join(__dirname, '../../../kanboard.db');
  db = new Database(resolvedPath);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');

  const schema = fs.readFileSync(path.join(__dirname, 'schema.sql'), 'utf-8');
  db.exec(schema);

  return db;
}

export function closeDb(): void {
  if (db) {
    db.close();
    db = undefined as any;
  }
}
```

```ts
export function requireAuth(req: Request, res: Response, next: NextFunction): void {
  if (!req.session.userId) {
    res.status(401).json({ error: 'Not authenticated' });
    return;
  }
  next();
}
```

- [ ] **Step 3: Create server/src/routes/auth.ts**

```ts
import { Router, Request, Response } from 'express';
import bcrypt from 'bcryptjs';
import Database from 'better-sqlite3';

export function authRoutes(db: Database.Database): Router {
  const router = Router();

  router.post('/login', (req: Request, res: Response) => {
    const { username, password } = req.body;

    if (!username || !password) {
      res.status(400).json({ error: 'Username and password required' });
      return;
    }

    const user = db.prepare('SELECT * FROM users WHERE username = ?').get(username) as any;

    if (!user || !bcrypt.compareSync(password, user.password_hash)) {
      res.status(401).json({ error: 'Invalid credentials' });
      return;
    }

    req.session.userId = user.id;
    req.session.username = user.username;
    req.session.role = user.role;

    res.json({ id: user.id, username: user.username, role: user.role });
  });

  router.post('/logout', (req: Request, res: Response) => {
    req.session.destroy((err) => {
      if (err) {
        res.status(500).json({ error: 'Logout failed' });
        return;
      }
      res.clearCookie('connect.sid');
      res.json({ ok: true });
    });
  });

  router.get('/me', (req: Request, res: Response) => {
    if (!req.session.userId) {
      res.status(401).json({ error: 'Not authenticated' });
      return;
    }
    res.json({
      id: req.session.userId,
      username: req.session.username,
      role: req.session.role,
    });
  });

  return router;
}
```

```ts
import { Router, Request, Response } from 'express';
import Database from 'better-sqlite3';

export function customerRoutes(db: Database.Database): Router {
  const router = Router();

  router.get('/', (_req: Request, res: Response) => {
    const customers = db.prepare(
      `SELECT c.*,
        (SELECT COUNT(*) FROM projects WHERE customer_id = c.id AND archived = 0) AS project_count,
        (SELECT COUNT(*) FROM blockers b JOIN projects p ON b.project_id = p.id
          WHERE p.customer_id = c.id AND b.resolved = 0) AS active_blockers
      FROM customers c ORDER BY c.name ASC`
    ).all();
    res.json(customers);
  });

  router.get('/:id', (req: Request, res: Response) => {
    const customer = db.prepare('SELECT * FROM customers WHERE id = ?').get(req.params.id);
    if (!customer) { res.status(404).json({ error: 'Customer not found' }); return; }

    const projects = db.prepare(
      `SELECT p.*,
        (SELECT COUNT(*) FROM tasks WHERE project_id = p.id AND stage = 'done') AS tasks_done,
        (SELECT COUNT(*) FROM tasks WHERE project_id = p.id) AS tasks_total
      FROM projects p WHERE p.customer_id = ? AND p.archived = 0
      ORDER BY p.updated_at DESC`
    ).all(req.params.id);

    res.json({ ...(customer as object), projects });
  });

  router.post('/', (req: Request, res: Response) => {
    const { name, email, phone, notes, color } = req.body;
    if (!name) { res.status(400).json({ error: 'Name is required' }); return; }

    const result = db.prepare(
      'INSERT INTO customers (name, email, phone, notes, color) VALUES (?, ?, ?, ?, ?)'
    ).run(name, email || null, phone || null, notes || null, color || '#6366f1');

    const customer = db.prepare('SELECT * FROM customers WHERE id = ?').get(result.lastInsertRowid);
    res.status(201).json(customer);
  });

  router.put('/:id', (req: Request, res: Response) => {
    const { name, email, phone, notes, color } = req.body;
    const existing = db.prepare('SELECT * FROM customers WHERE id = ?').get(req.params.id) as any;
    if (!existing) { res.status(404).json({ error: 'Customer not found' }); return; }

    db.prepare(
      'UPDATE customers SET name = ?, email = ?, phone = ?, notes = ?, color = ? WHERE id = ?'
    ).run(
      name ?? existing.name, email ?? existing.email, phone ?? existing.phone,
      notes ?? existing.notes, color ?? existing.color, req.params.id
    );

    const customer = db.prepare('SELECT * FROM customers WHERE id = ?').get(req.params.id);
    res.json(customer);
  });

  router.delete('/:id', (req: Request, res: Response) => {
    const existing = db.prepare('SELECT * FROM customers WHERE id = ?').get(req.params.id);
    if (!existing) { res.status(404).json({ error: 'Customer not found' }); return; }
    db.prepare('DELETE FROM customers WHERE id = ?').run(req.params.id);
    res.json({ ok: true });
  });

  return router;
}
```

```ts
      'INSERT INTO tasks (project_id, title, description, stage, sort_order) VALUES (?, ?, ?, ?, ?)'

    res.status(201).json(db.prepare('SELECT * FROM tasks WHERE id = ?').get(result.lastInsertRowid));
  });

  router.put('/:taskId', (req: Request, res: Response) => {
```

```ts
    const { resolved, description } = req.body;

        resolved ? 1 : 0, resolved ? new Date().toISOString() : null, req.params.blockerId
      );

      db.prepare('UPDATE blockers SET description = ? WHERE id = ?').run(description, req.params.blockerId);
    }
```

```ts
export function createInAppNotification(db: Database.Database, params: {
  projectId: number; type: string; message: string; userId?: number | null;
}): void {
  if (params.userId) {
    db.prepare('INSERT INTO notifications (user_id, project_id, type, message) VALUES (?, ?, ?, ?)').run(
      params.userId, params.projectId, params.type, params.message
    );
  } else {
    const users = db.prepare('SELECT id FROM users').all() as { id: number }[];
    const insert = db.prepare('INSERT INTO notifications (user_id, project_id, type, message) VALUES (?, ?, ?, ?)');
    for (const u of users) insert.run(u.id, params.projectId, params.type, params.message);
  }
}
```

```ts
export function notificationRoutes(db: Database.Database): Router {
  const router = Router();

  router.get('/', (req: Request, res: Response) => {
    const { type } = req.query;
    let sql = `SELECT n.*, p.title AS project_title, c.name AS customer_name
      FROM notifications n LEFT JOIN projects p ON n.project_id = p.id
      LEFT JOIN customers c ON p.customer_id = c.id WHERE n.user_id = ?`;
    const params: any[] = [req.session.userId];
    if (type) { sql += ' AND n.type = ?'; params.push(type); }
    sql += ' ORDER BY n.created_at DESC LIMIT 100';
    res.json(db.prepare(sql).all(...params));
  });

  router.get('/unread-count', (req: Request, res: Response) => {
    const result = db.prepare('SELECT COUNT(*) AS count FROM notifications WHERE user_id = ? AND read = 0').get(req.session.userId) as { count: number };
    res.json({ count: result.count });
  });

  router.patch('/:id/read', (req: Request, res: Response) => {
    db.prepare('UPDATE notifications SET read = 1 WHERE id = ? AND user_id = ?').run(req.params.id, req.session.userId);
    res.json({ ok: true });
  });

  router.post('/read-all', (req: Request, res: Response) => {
    db.prepare('UPDATE notifications SET read = 1 WHERE user_id = ? AND read = 0').run(req.session.userId);
    res.json({ ok: true });
  });

  return router;
}
```

```ts
export function dashboardRoutes(db: Database.Database): Router {
  const router = Router();

  router.get('/', (_req: Request, res: Response) => {
    const activeProjects = (db.prepare("SELECT COUNT(*) AS count FROM projects WHERE archived = 0 AND stage != 'done'").get() as any).count;
    const blockedCount = (db.prepare('SELECT COUNT(*) AS count FROM blockers WHERE resolved = 0').get() as any).count;
    const dueThisWeek = (db.prepare("SELECT COUNT(*) AS count FROM projects WHERE due_date BETWEEN date('now') AND date('now', '+7 days') AND archived = 0 AND stage != 'done'").get() as any).count;
    const completedThisMonth = (db.prepare("SELECT COUNT(*) AS count FROM projects WHERE stage = 'done' AND updated_at >= date('now', 'start of month')").get() as any).count;

    const activeBlockers = db.prepare(
      `SELECT b.*, p.title AS project_title, c.name AS customer_name, p.id AS project_id,
        CAST(julianday('now') - julianday(b.created_at) AS INTEGER) AS days_blocked
      FROM blockers b JOIN projects p ON b.project_id = p.id JOIN customers c ON p.customer_id = c.id
      WHERE b.resolved = 0 ORDER BY b.created_at ASC`
    ).all();

    const dueSoon = db.prepare(
      `SELECT p.*, c.name AS customer_name, c.color AS customer_color
      FROM projects p JOIN customers c ON p.customer_id = c.id
      WHERE p.due_date BETWEEN date('now') AND date('now', '+7 days') AND p.archived = 0 AND p.stage != 'done'
      ORDER BY p.due_date ASC`
    ).all();

    res.json({
      stats: { active_projects: activeProjects, blocked: blockedCount, due_this_week: dueThisWeek, completed_this_month: completedThisMonth },
      active_blockers: activeBlockers,
      due_soon: dueSoon,
      recent_activity: getRecentActivity(db, 20),
    });
  });

  return router;
}
```

```ts
import cron from 'node-cron';
import Database from 'better-sqlite3';
import { createInAppNotification, sendSlackNotification } from './notify.js';

export function startCronJobs(db: Database.Database): void {
  const slackUrl = process.env.SLACK_WEBHOOK_URL || '';

  cron.schedule('0 8 * * *', async () => {
    console.log('[cron] Running overdue scan...');

    const overdue = db.prepare(
      `SELECT p.*, c.name AS customer_name FROM projects p JOIN customers c ON p.customer_id = c.id
      WHERE p.due_date < date('now') AND p.stage NOT IN ('done') AND p.archived = 0`
    ).all() as any[];

    for (const p of overdue) {
      const msg = `Overdue: "${p.title}" for ${p.customer_name} was due ${p.due_date}`;
      createInAppNotification(db, { projectId: p.id, type: 'overdue', message: msg });
      if (slackUrl) await sendSlackNotification(slackUrl, msg);
    }

    const stale = db.prepare(
      `SELECT b.*, p.title AS project_title, c.name AS customer_name FROM blockers b
      JOIN projects p ON b.project_id = p.id JOIN customers c ON p.customer_id = c.id
      WHERE b.resolved = 0 AND b.created_at <= datetime('now', '-3 days')`
    ).all() as any[];

    for (const b of stale) {
      const msg = `Unresolved 3+ days: "${b.description}" on ${b.project_title} (${b.customer_name})`;
      createInAppNotification(db, { projectId: b.project_id, type: 'blocker', message: msg });
      if (slackUrl) await sendSlackNotification(slackUrl, msg);
    }

    console.log(`[cron] ${overdue.length} overdue, ${stale.length} stale blockers`);
  });
}
```
